class Point3D: # 定义点类
def __init__(self, x=0.0, y=0.0, z=0.0, w = 1.0):
self.x, self.y, self.z, self.w = x, y, z, w
class Vector3D: # 定义向量类
def __init__(self, dx = 0.0, dy = 0.0, dz = 0.0, dw = 0.0):
self.dx, self.dy, self.dz, self.dw = dx, dy, dz, dw
